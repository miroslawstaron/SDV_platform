cmake_minimum_required(VERSION 3.16)
project(sdv_adapter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(SilKit CONFIG REQUIRED)
find_package(vsomeip3 CONFIG REQUIRED)
find_package(Protobuf REQUIRED)
find_package(gRPC CONFIG REQUIRED)
set(PROTO ${CMAKE_CURRENT_SOURCE_DIR}/../contracts/adapter.proto)
add_custom_command(OUTPUT adapter.pb.cc adapter.pb.h adapter.grpc.pb.cc adapter.grpc.pb.h COMMAND protobuf::protoc ARGS --cpp_out=${CMAKE_CURRENT_BINARY_DIR} --grpc_out=${CMAKE_CURRENT_BINARY_DIR} --plugin=protoc-gen-grpc=$<TARGET_FILE:gRPC::grpc_cpp_plugin> -I ${CMAKE_CURRENT_SOURCE_DIR}/../contracts ${PROTO} DEPENDS ${PROTO})
add_library(adapter_proto adapter.pb.cc adapter.grpc.pb.cc)
target_include_directories(adapter_proto PUBLIC ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(adapter_proto PUBLIC protobuf::libprotobuf gRPC::grpc++)
add_executable(sdv-adapter adapter.cpp)
target_link_libraries(sdv-adapter PRIVATE adapter_proto SilKit::SilKit vsomeip3 pthread)
add_executable(example-controller ../examples/cpp/controller.cpp)
add_executable(sdk-smoke sdk-smoke.cpp)
target_link_libraries(sdk-smoke PRIVATE adapter_proto)
